1. 系统概述 (System Overview)

1.1 定义与愿景

环境管理模块 (Environment Manager) 是重明架构中负责 "配" (Configure) 和 "切" (Switch) 的基础组件。它像一个智能的环境调度员，管理测试执行所需的各类环境配置，支持一键切换开发/测试/预发/生产环境。

其核心设计哲学是 "Config-as-Code, Environment-Agnostic" (配置即代码，环境无关)：
- 配置集中：统一管理所有环境配置
- 一键切换：快速切换目标环境
- 变量注入：自动注入环境变量到测试用例
- 敏感加密：加密存储敏感信息 (密码、Token)

1.2 核心职责
- 环境定义 (Environment Definition)：定义各环境的配置结构
- 配置管理 (Config Management)：集中管理 URL、凭证、参数
- 环境切换 (Environment Switch)：一键切换目标环境
- 变量注入 (Variable Injection)：运行时注入环境变量
- 敏感加密 (Secret Encryption)：安全存储敏感配置
- 健康检查 (Health Check)：验证环境可用性

4. 配置加载层 (Config Loader Layer)

4.1 配置加载器

class ConfigLoader:
    def __init__(self, config_path: str = "environments.yaml"):
        self.config_path = config_path
        self.environments = {}
    
    def load(self) -> dict[str, Environment]:
        # 1. 加载主配置文件
        with open(self.config_path, 'r') as f:
            config = yaml.safe_load(f)
        
        # 2. 加载环境变量覆盖
        overrides = self._load_env_overrides()
        
        # 3. 解析每个环境
        for env_key, env_config in config['environments'].items():
            # 应用覆盖
            if env_key in overrides:
                env_config = self._merge_config(env_config, overrides[env_key])
            
            self.environments[env_key] = Environment(
                key=env_key,
                **env_config
            )
        
        return self.environments
    
    def _load_env_overrides(self) -> dict:
        """从环境变量加载覆盖配置"""
        overrides = {}
        prefix = "CHONGMING_ENV_"
        
        for key, value in os.environ.items():
            if key.startswith(prefix):
                # CHONGMING_ENV_DEV_API_URL → dev.urls.api
                parts = key[len(prefix):].lower().split('_')
                env_key = parts[0]
                config_path = '.'.join(parts[1:])
                
                if env_key not in overrides:
                    overrides[env_key] = {}
                self._set_nested(overrides[env_key], config_path, value)
        
        return overrides

5. 敏感信息管理 (Secret Management)

5.1 加密存储

class SecretManager:
    def __init__(self, key_path: str = ".chongming_key"):
        self.cipher = self._load_or_create_key(key_path)
        self.secrets = {}
    
    def encrypt(self, plaintext: str) -> str:
        """加密敏感信息"""
        encrypted = self.cipher.encrypt(plaintext.encode())
        return base64.b64encode(encrypted).decode()
    
    def decrypt(self, ciphertext: str) -> str:
        """解密敏感信息"""
        encrypted = base64.b64decode(ciphertext.encode())
        return self.cipher.decrypt(encrypted).decode()
    
    def load_secrets(self, secrets_file: str = "secrets.enc"):
        """加载加密的密钥文件"""
        with open(secrets_file, 'r') as f:
            encrypted_data = json.load(f)
        
        for key, value in encrypted_data.items():
            self.secrets[key] = self.decrypt(value)
    
    def get(self, key: str) -> str:
        return self.secrets.get(key)

6. 环境切换层 (Environment Switcher)

6.1 环境切换器

class EnvironmentSwitcher:
    def __init__(self, config_loader: ConfigLoader):
        self.loader = config_loader
        self.current_env = None
    
    def switch(self, env_key: str) -> Environment:
        """切换到指定环境"""
        environments = self.loader.load()
        
        if env_key not in environments:
            raise EnvironmentNotFoundError(f"环境 '{env_key}' 不存在")
        
        target_env = environments[env_key]
        
        # 1. 验证环境权限
        self._validate_permissions(target_env)
        
        # 2. 健康检查
        if not self._health_check(target_env):
            raise EnvironmentUnavailableError(f"环境 '{env_key}' 不可用")
        
        # 3. 切换
        self.current_env = target_env
        
        # 4. 通知观察者
        self._notify_switch(target_env)
        
        return target_env
    
    def get_current(self) -> Environment:
        return self.current_env
    
    def _validate_permissions(self, env: Environment):
        # 生产环境需要额外确认
        if env.key == "prod" and not os.getenv("ALLOW_PROD_ACCESS"):
            raise PermissionError("访问生产环境需要设置 ALLOW_PROD_ACCESS=true")

7. 变量注入层 (Variable Injection)

7.1 变量解析器

class VariableResolver:
    def __init__(self, env: Environment, secrets: SecretManager):
        self.env = env
        self.secrets = secrets
        self.pattern = re.compile(r'\$\{(\w+(?:\.\w+)*)\}')
    
    def resolve(self, text: str) -> str:
        """解析并替换变量引用"""
        def replacer(match):
            var_path = match.group(1)
            return self._get_value(var_path)
        
        return self.pattern.sub(replacer, text)
    
    def _get_value(self, var_path: str) -> str:
        # 处理 secrets 引用
        if var_path.startswith("secrets."):
            secret_key = var_path[8:]  # 去掉 "secrets." 前缀
            return self.secrets.get(secret_key)
        
        # 处理环境变量引用
        if var_path.startswith("env."):
            env_key = var_path[4:]
            return os.getenv(env_key, "")
        
        # 处理环境配置引用
        return self._get_nested(self.env, var_path)
    
    def inject_to_tc(self, tc_ir: TCIR) -> TCIR:
        """将环境变量注入到测试用例"""
        # 替换 URL
        for step in tc_ir.steps:
            if step.target:
                step.target = self.resolve(step.target)
            if step.value:
                step.value = self.resolve(step.value)
        
        return tc_ir

8. 健康检查层 (Health Checker)

8.1 健康检查器

class HealthChecker:
    def __init__(self):
        self.checks = {
            "web": self._check_web,
            "api": self._check_api,
            "database": self._check_database
        }
    
    async def check_all(self, env: Environment) -> HealthReport:
        results = {}
        
        for check_name, check_func in self.checks.items():
            try:
                result = await check_func(env)
                results[check_name] = HealthResult(
                    status="healthy" if result else "unhealthy",
                    latency_ms=result.latency if result else None
                )
            except Exception as e:
                results[check_name] = HealthResult(
                    status="error",
                    error=str(e)
                )
        
        return HealthReport(
            environment=env.key,
            timestamp=datetime.now(),
            overall_status=self._calculate_overall(results),
            details=results
        )
    
    async def _check_web(self, env: Environment) -> CheckResult:
        start = time.time()
        async with httpx.AsyncClient() as client:
            response = await client.get(env.urls.web, timeout=10)
            latency = (time.time() - start) * 1000
            return CheckResult(
                success=response.status_code == 200,
                latency=latency
            )
    
    async def _check_api(self, env: Environment) -> CheckResult:
        start = time.time()
        async with httpx.AsyncClient() as client:
            response = await client.get(f"{env.urls.api}/health", timeout=10)
            latency = (time.time() - start) * 1000
            return CheckResult(
                success=response.status_code == 200,
                latency=latency
            )

9. 环境隔离与限制 (Environment Isolation)

9.1 生产环境保护

class ProductionGuard:
    """生产环境保护机制"""
    
    DANGEROUS_OPERATIONS = [
        "delete", "drop", "truncate", "update", "insert",
        "click_delete", "click_submit", "fill_form"
    ]
    
    def validate_operation(self, env: Environment, operation: str) -> bool:
        if not env.read_only:
            return True
        
        # 检查是否为危险操作
        if any(op in operation.lower() for op in self.DANGEROUS_OPERATIONS):
            raise ProductionGuardError(
                f"禁止在生产环境执行危险操作: {operation}"
            )
        
        # 检查是否在允许列表
        if env.allowed_operations:
            if operation not in env.allowed_operations:
                raise ProductionGuardError(
                    f"操作 '{operation}' 不在生产环境允许列表中"
                )
        
        return True
