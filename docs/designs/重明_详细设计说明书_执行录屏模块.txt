1. 模块概述 (System Overview)

1.1 定义与愿景


执行录屏模块 (Execution Recorder) 是重明架构中负责 "录" (Record) 和 "播" (Playback) 测试执行过程的核心可视化组件。它像一个智能摄影师，自动捕捉测试执行的每一个关键帧，生成可回溯、可分析的执行轨迹。

其核心设计哲学是 "Capture Everything, Replay Anything" (全程捕获，随时回放)：
- 智能捕获：自动识别关键操作时机并截屏
- 元素标注：在截图上高亮显示操作目标
- 时间线回放：按步骤顺序重现执行过程
- 可视化分析：直观展示测试流程和结果

### 1.2 核心价值

1. **缺陷定位可视化** - 通过录屏快速定位失败步骤
2. **执行过程可追溯** - 完整记录测试执行轨迹
3. **团队协作增强** - 可分享的可视化测试报告
4. **AI 分析支撑** - 为缺陷分析智能体提供视觉上下文

### 4.2 元素标注器

#### 4.2.1 边框绘制算法

```typescript
/**
 * 元素标注器
 */
class ElementAnnotator {
  private colors = DEFAULT_COLORS;
  
  /**
   * 在截图上绘制元素标注
   */
  async annotate(
    screenshot: string,
    elements: ElementPosition[],
    options?: AnnotationOptions
  ): Promise<string> {
    const { width, height, jimpImage } = await loadImage(screenshot);
    const overlay = await this.createOverlay(elements, width, height, options);
    
    // 合成原图与标注层
    const result = await jimpImage.composite(overlay, 0, 0, {
      mode: BLEND_SOURCE_OVER,
      opacitySource: 1,
      opacityDest: 1
    });
    
    return await result.getBase64Async(MIME_JPEG);
  }
  
  /**
   * 创建标注覆盖层
   */
  private async createOverlay(
    elements: ElementPosition[],
    width: number,
    height: number,
    options?: AnnotationOptions
  ): Promise<JimpImage> {
    const padding = options?.padding ?? 5;
    const thickness = options?.borderThickness ?? 2;
    const overlay = new Jimp(width, height, 0x00000000);
    
    for (let i = 0; i < elements.length; i++) {
      const elem = elements[i];
      const color = this.colors[i % this.colors.length];
      
      // 计算带边距的矩形
      const rect = {
        left: Math.max(0, elem.rect.left - padding),
        top: Math.max(0, elem.rect.top - padding),
        width: Math.min(width - elem.rect.left, elem.rect.width + 2 * padding),
        height: Math.min(height - elem.rect.top, elem.rect.height + 2 * padding)
      };
      
      // 绘制边框
      this.drawBorder(overlay, rect, color.rect, thickness);
      
      // 绘制索引标签
      if (typeof elem.indexId === 'number') {
        this.drawIndexLabel(overlay, elem.indexId, rect, color);
      }
    }
    
    return overlay;
  }
  
  /**
   * 绘制矩形边框
   */
  private drawBorder(
    image: JimpImage, 
    rect: ElementRect, 
    color: number, 
    thickness: number
  ): void {
    const { left, top, width, height } = rect;
    
    image.scan(left, top, width, height, (x, y, idx) => {
      const isOnBorder = 
        (x >= left && x < left + thickness) ||
        (x <= left + width - 1 && x > left + width - thickness) ||
        (y >= top && y < top + thickness) ||
        (y <= top + height - 1 && y > top + height - thickness);
      
      if (isOnBorder) {
        image.bitmap.data[idx + 0] = (color >> 24) & 0xff; // R
        image.bitmap.data[idx + 1] = (color >> 16) & 0xff; // G
        image.bitmap.data[idx + 2] = (color >> 8) & 0xff;  // B
        image.bitmap.data[idx + 3] = color & 0xff;         // A
      }
    });
  }
}
```

```typescript
/**
 * 计算标签位置，避免与其他元素重叠
 */
private calculateLabelPosition(
  elementIndex: number,
  currentRect: ElementRect,
  previousElements: ElementPosition[],
  canvasWidth: number,
  canvasHeight: number
): { x: number; y: number } {
  const labelWidth = 8 * String(elementIndex).length + 5;
  const labelHeight = 16;
  
  // 候选位置 (按优先级)
  const candidates = [
    { x: currentRect.left - labelWidth, y: currentRect.top + currentRect.height / 2 - 8 },  // 左侧
    { x: currentRect.left, y: currentRect.top - labelHeight - 2 },                           // 上方
    { x: currentRect.left, y: currentRect.top + currentRect.height + 2 },                    // 下方
    { x: currentRect.left + currentRect.width + 2, y: currentRect.top },                     // 右侧
    { x: currentRect.left, y: currentRect.top + 2 }                                          // 内部左上
  ];
  
  // 检查函数：是否与其他元素重叠
  const overlapsWithPrevious = (x: number, y: number): boolean => {
    return previousElements.some(elem => 
      x < elem.rect.left + elem.rect.width &&
      x + labelWidth > elem.rect.left &&
      y < elem.rect.top + elem.rect.height &&
      y + labelHeight > elem.rect.top
    );
  };
  
  // 检查函数：是否在画布内
  const isWithinCanvas = (x: number, y: number): boolean => {
    return x >= 0 && x + labelWidth <= canvasWidth &&
           y >= 0 && y + labelHeight <= canvasHeight;
  };
  
  // 找到第一个不重叠且在画布内的位置
  for (const pos of candidates) {
    if (!overlapsWithPrevious(pos.x, pos.y) && isWithinCanvas(pos.x, pos.y)) {
      return pos;
    }
  }
  
  // 默认返回第一个候选位置
  return candidates[0];
}
```

### 4.3 时间线管理器

```typescript
/**
 * 时间线管理器
 */
class TimelineManager {
  private frames: TimelineFrame[] = [];
  
  /**
   * 添加帧到时间线
   */
  addFrame(frame: RecorderFrame, taskInfo: TaskInfo): void {
    this.frames.push({
      ...frame,
      taskId: taskInfo.id,
      taskType: taskInfo.type,
      taskSubType: taskInfo.subType,
      taskStatus: taskInfo.status
    });
  }
  
  /**
   * 获取按时间排序的帧列表
   */
  getTimeline(): TimelineFrame[] {
    return [...this.frames].sort((a, b) => a.ts - b.ts);
  }
  
  /**
   * 获取指定任务的所有帧
   */
  getFramesByTask(taskId: string): TimelineFrame[] {
    return this.frames.filter(f => f.taskId === taskId);
  }
  
  /**
   * 计算总执行时长
   */
  getTotalDuration(): number {
    if (this.frames.length === 0) return 0;
    const sorted = this.getTimeline();
    return sorted[sorted.length - 1].ts - sorted[0].ts;
  }
  
  /**
   * 导出时间线数据
   */
  export(): TimelineExport {
    return {
      frames: this.getTimeline(),
      totalDuration: this.getTotalDuration(),
      frameCount: this.frames.length
    };
  }
}
```

### 4.4 回放播放器

```typescript
/**
 * 录制回放播放器
 */
class RecordingPlayer {
  private currentIndex: number = 0;
  private isPlaying: boolean = false;
  private playbackSpeed: number = 1.0;
  private frames: TimelineFrame[] = [];
  
  /**
   * 加载录制数据
   */
  load(recording: ExecutionRecording): void {
    this.frames = this.extractFrames(recording);
    this.currentIndex = 0;
  }
  
  /**
   * 从录制数据中提取所有帧
   */
  private extractFrames(recording: ExecutionRecording): TimelineFrame[] {
    const frames: TimelineFrame[] = [];
    
    for (const execution of recording.executions) {
      for (const task of execution.tasks) {
        if (task.recorder) {
          for (const frame of task.recorder) {
            frames.push({
              ...frame,
              taskName: execution.name,
              taskType: task.type,
              taskSubType: task.subType,
              taskStatus: task.status
            });
          }
        }
      }
    }
    
    return frames.sort((a, b) => a.ts - b.ts);
  }
  
  /**
   * 播放/暂停切换
   */
  togglePlay(): void {
    this.isPlaying = !this.isPlaying;
    if (this.isPlaying) {
      this.playLoop();
    }
  }
  
  /**
   * 跳转到指定帧
   */
  seekTo(index: number): void {
    this.currentIndex = Math.max(0, Math.min(index, this.frames.length - 1));
  }
  
  /**
   * 上一帧
   */
  previousFrame(): void {
    if (this.currentIndex > 0) {
      this.currentIndex--;
    }
  }
  
  /**
   * 下一帧
   */
  nextFrame(): void {
    if (this.currentIndex < this.frames.length - 1) {
      this.currentIndex++;
    }
  }
  
  /**
   * 获取当前帧
   */
  getCurrentFrame(): TimelineFrame | null {
    return this.frames[this.currentIndex] || null;
  }
  
  /**
   * 设置播放速度
   */
  setSpeed(speed: number): void {
    this.playbackSpeed = Math.max(0.25, Math.min(4.0, speed));
  }
  
  /**
   * 播放循环
   */
  private async playLoop(): Promise<void> {
    while (this.isPlaying && this.currentIndex < this.frames.length - 1) {
      const currentFrame = this.frames[this.currentIndex];
      const nextFrame = this.frames[this.currentIndex + 1];
      const delay = (nextFrame.ts - currentFrame.ts) / this.playbackSpeed;
      
      await sleep(Math.max(50, delay));
      
      if (this.isPlaying) {
        this.currentIndex++;
      }
    }
    
    this.isPlaying = false;
  }
}
```

```typescript
/**
 * 点击涟漪特效配置
 */
interface ClickRippleConfig {
  /** 涟漪中心点 */
  center: [number, number];
  /** 涟漪颜色 (RGBA) */
  color: string;
  /** 最大半径 (像素) */
  maxRadius: number;
  /** 动画持续时间 (毫秒) */
  duration: number;
  /** 涟漪圈数 */
  rings: number;
}

/**
 * 点击涟漪渲染器
 */
class ClickRippleRenderer {
  private defaultConfig: ClickRippleConfig = {
    center: [0, 0],
    color: 'rgba(255, 87, 34, 0.8)',  // 橙红色
    maxRadius: 40,
    duration: 600,
    rings: 3
  };

  /**
   * 在截图上绘制点击涟漪效果 (静态帧)
   */
  async renderRipple(
    screenshot: string,
    clickPoint: [number, number],
    config?: Partial<ClickRippleConfig>
  ): Promise<string> {
    const cfg = { ...this.defaultConfig, ...config, center: clickPoint };
    const { width, height, jimpImage } = await loadImage(screenshot);
    
    // 绘制多层涟漪圈
    for (let ring = 0; ring < cfg.rings; ring++) {
      const radius = cfg.maxRadius * (ring + 1) / cfg.rings;
      const opacity = 1 - (ring / cfg.rings) * 0.7;
      await this.drawRippleRing(jimpImage, cfg.center, radius, cfg.color, opacity);
    }
    
    // 绘制中心点标记
    await this.drawCenterDot(jimpImage, cfg.center, 6, cfg.color);
    
    return await jimpImage.getBase64Async(MIME_JPEG);
  }

  /**
   * 绘制单个涟漪圈
   */
  private async drawRippleRing(
    image: JimpImage,
    center: [number, number],
    radius: number,
    color: string,
    opacity: number
  ): Promise<void> {
    const [cx, cy] = center;
    const thickness = 3;
    
    // 扫描绘制圆环
    for (let angle = 0; angle < 360; angle += 1) {
      const radian = (angle * Math.PI) / 180;
      for (let r = radius - thickness; r <= radius; r++) {
        const x = Math.round(cx + r * Math.cos(radian));
        const y = Math.round(cy + r * Math.sin(radian));
        this.setPixelWithOpacity(image, x, y, color, opacity);
      }
    }
  }

  /**
   * 绘制中心实心圆点
   */
  private async drawCenterDot(
    image: JimpImage,
    center: [number, number],
    radius: number,
    color: string
  ): Promise<void> {
    const [cx, cy] = center;
    
    for (let x = cx - radius; x <= cx + radius; x++) {
      for (let y = cy - radius; y <= cy + radius; y++) {
        const distance = Math.sqrt((x - cx) ** 2 + (y - cy) ** 2);
        if (distance <= radius) {
          this.setPixelWithOpacity(image, Math.round(x), Math.round(y), color, 1);
        }
      }
    }
```

```typescript
  /**
   * 绘制端点标记
   */
  private async drawEndpoint(
    image: JimpImage,
    point: [number, number],
    color: string,
    type: 'start' | 'end'
  ): Promise<void> {
    const radius = type === 'start' ? 8 : 10;
    // 绘制实心圆
    await this.drawFilledCircle(image, point, radius, color);
    // 绘制白色内圈
    await this.drawFilledCircle(image, point, radius - 3, 'rgba(255, 255, 255, 1)');
  }

  /**
   * 使用 Bresenham 算法绘制线条
   */
  private async drawLine(
    image: JimpImage,
    from: [number, number],
    to: [number, number],
    color: string,
    thickness: number,
    style: 'solid' | 'dashed' | 'dotted'
  ): Promise<void> {
    const [x1, y1] = from;
    const [x2, y2] = to;
    
    const dx = Math.abs(x2 - x1);
    const dy = Math.abs(y2 - y1);
    const sx = x1 < x2 ? 1 : -1;
    const sy = y1 < y2 ? 1 : -1;
    
    let err = dx - dy;
    let x = x1, y = y1;
    let dashCounter = 0;
    
    while (true) {
      // 根据样式决定是否绘制
      const shouldDraw = 
        style === 'solid' ||
        (style === 'dashed' && (dashCounter % 12) < 8) ||
        (style === 'dotted' && (dashCounter % 6) < 3);
      
      if (shouldDraw) {
        // 绘制粗线 (以当前点为中心的小圆)
        for (let tx = -thickness/2; tx <= thickness/2; tx++) {
          for (let ty = -thickness/2; ty <= thickness/2; ty++) {
            if (Math.sqrt(tx*tx + ty*ty) <= thickness/2) {
              this.setPixel(image, x + tx, y + ty, color);
            }
          }
        }
      }
      
      if (x === x2 && y === y2) break;
      
      const e2 = 2 * err;
      if (e2 > -dy) { err -= dy; x += sx; }
      if (e2 < dx) { err += dx; y += sy; }
      dashCounter++;
    }
  }

  /**
   * 绘制箭头
   */
  private async drawArrowHead(
    image: JimpImage,
    from: [number, number],
    to: [number, number],
    color: string
  ): Promise<void> {
    const arrowLength = 20;
    const arrowAngle = Math.PI / 6; // 30度
    
    const angle = Math.atan2(to[1] - from[1], to[0] - from[0]);
    
    const arrow1 = [
      to[0] - arrowLength * Math.cos(angle - arrowAngle),
      to[1] - arrowLength * Math.sin(angle - arrowAngle)
    ] as [number, number];
    
    const arrow2 = [
      to[0] - arrowLength * Math.cos(angle + arrowAngle),
      to[1] - arrowLength * Math.sin(angle + arrowAngle)
    ] as [number, number];
    
    await this.drawLine(image, arrow1, to, color, 3, 'solid');
    await this.drawLine(image, arrow2, to, color, 3, 'solid');
  }
}
```

```typescript
class InteractionEffectManager {
  private clickRenderer = new ClickRippleRenderer();
  private dragRenderer = new DragTrailRenderer();

   * 根据操作类型自动添加对应特效
   */

    screenshot: string,
    action: TaskRecord
  ): Promise<string> {
    const { subType, param, output } = action;
    
    switch (subType) {
      case 'Tap':
      case 'Click':
        const clickPoint = output?.element?.center || this.extractCenterFromParam(param);
        if (clickPoint) {
          return await this.clickRenderer.renderRipple(screenshot, clickPoint);
        }
        break;
        
      case 'DoubleClick':
        const dblClickPoint = output?.element?.center || this.extractCenterFromParam(param);

          return await this.clickRenderer.renderRipple(screenshot, dblClickPoint, {
            rings: 5,
            color: 'rgba(233, 30, 99, 0.8)'
          });
        }
        break;
        
      case 'DragAndDrop':
        const fromPoint = this.extractCenterFromParam(param?.from);
        const toPoint = this.extractCenterFromParam(param?.to);
        if (fromPoint && toPoint) {
          return await this.dragRenderer.renderTrail(screenshot, fromPoint, toPoint);
        }
        break;
        
      case 'Scroll':
        const scrollCenter = output?.element?.center || [screenshot.width / 2, screenshot.height / 2];
        return await this.scrollRenderer.renderIndicator(screenshot, {
          center: scrollCenter as [number, number],
          direction: param?.direction || 'down',
          distance: param?.distance || 200,
          color: 'rgba(156, 39, 176, 0.8)'
        });
        
      case 'Input':
        const inputRect = output?.element?.rect;
        if (inputRect) {
          return await this.inputRenderer.renderHighlight(screenshot, {
            rect: inputRect,
            inputText: param?.value || '',
            color: 'rgba(255, 193, 7, 0.8)',
            showCursor: true,
            showTextBubble: true
          });
        }
        break;
        
      case 'Hover':
        const hoverPoint = output?.element?.center;
        if (hoverPoint) {
          return await this.clickRenderer.renderRipple(screenshot, hoverPoint, {
            color: 'rgba(100, 181, 246, 0.5)',
            maxRadius: 30,
            rings: 2
          });
        }
        break;
    }
    
    return screenshot; // 无特效则返回原图
  }

  /**
   * 从参数中提取中心点
   */
  private extractCenterFromParam(param: any): [number, number] | null {
    if (!param) return null;
    
    if (param.center) return param.center;
    
    if (param.rect) {
      return [
        param.rect.left + param.rect.width / 2,
        param.rect.top + param.rect.height / 2
      ];
    }
    
    if (param.bbox && param.bbox.length === 4) {
      return [
        (param.bbox[0] + param.bbox[2]) / 2,
        (param.bbox[1] + param.bbox[3]) / 2
      ];
    }
    
    return null;
  }
```

```typescript
class RecordingSerializer {

   * 序列化为 JSON 字符串

  /**
   * 自定义序列化处理
   */
  private replacer(key: string, value: any): any {
    // 处理特殊对象类型

      return '[Page object]';
    }
    if (value?.constructor?.name === 'Browser') {
      return '[Browser object]';
    }
    if (typeof value?.toSerializable === 'function') {
      return value.toSerializable();
    }
    return value;
  }
  
  /**
   * 从 JSON 字符串反序列化
   */
  deserialize(json: string): ExecutionRecording {
    return JSON.parse(json, this.reviver);
  }
  
  /**
   * 自定义反序列化处理
   */
  private reviver(key: string, value: any): any {
    // 恢复截图数据对象
    if (key === 'screenshot' && ScreenshotData.isSerializedData(value)) {
      return ScreenshotData.fromSerializedData(value);
    }
    return value;
  }
}
```

## 8. 性能优化

### 8.1 增量存储

```typescript
/**
 * 增量帧存储
 * 仅存储与上一帧的差异部分
 */
class IncrementalFrameStore {
  private baseFrame: string | null = null;
  private diffFrames: DiffFrame[] = [];
  
  /**
   * 添加帧 (自动计算差异)
   */
  async addFrame(screenshot: string): Promise<void> {
    if (!this.baseFrame) {
      this.baseFrame = screenshot;
      return;
    }
    
    // 计算与上一帧的差异
    const diff = await this.computeDiff(this.getLastFrame(), screenshot);
    
    // 如果差异过大，存储为新的基准帧
    if (diff.changeRatio > 0.5) {
      this.diffFrames.push({ type: 'keyframe', data: screenshot });
    } else {
      this.diffFrames.push({ type: 'diff', data: diff.patch });
    }
  }
  
  /**
   * 重建指定索引的帧
   */
  async reconstructFrame(index: number): Promise<string> {
    let frame = this.baseFrame!;
    
    for (let i = 0; i <= index && i < this.diffFrames.length; i++) {
      const diff = this.diffFrames[i];
      if (diff.type === 'keyframe') {
        frame = diff.data;
      } else {
        frame = await this.applyDiff(frame, diff.data);
      }
    }
    
    return frame;
  }
}
```

### 8.2 懒加载与缓存

```typescript
/**
 * 帧懒加载管理器
 */
class LazyFrameLoader {
  private cache = new LRUCache<string, string>({ max: 50 });
  private loadingPromises = new Map<string, Promise<string>>();
  
  /**
   * 获取帧 (带缓存)
   */
  async getFrame(frameId: string): Promise<string> {
    // 检查缓存
    const cached = this.cache.get(frameId);
    if (cached) return cached;
    
    // 检查是否正在加载
    const loading = this.loadingPromises.get(frameId);
    if (loading) return loading;
    
    // 开始加载
    const promise = this.loadFrame(frameId);
    this.loadingPromises.set(frameId, promise);
    
    try {
      const frame = await promise;
      this.cache.set(frameId, frame);
      return frame;
    } finally {
      this.loadingPromises.delete(frameId);
    }
  }
  
  /**
   * 预加载附近帧
   */
  async preloadNearby(currentIndex: number, frames: TimelineFrame[]): Promise<void> {
    const preloadRange = 3;
    const start = Math.max(0, currentIndex - preloadRange);
    const end = Math.min(frames.length - 1, currentIndex + preloadRange);
    
    for (let i = start; i <= end; i++) {
      if (i !== currentIndex) {
        this.getFrame(frames[i].id); // 异步预加载，不等待
      }
    }
  }
}
```
