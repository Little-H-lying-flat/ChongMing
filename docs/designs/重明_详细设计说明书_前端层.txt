1. 系统概述 (System Overview)

1.1 定义与愿景

前端层 (Frontend Layer) 是重明架构中负责 "展" (Present) 和 "交" (Interact) 的核心组件。它是用户与重明平台交互的唯一入口，承载着测试用例管理、执行监控、报告可视化等全部用户界面功能。

其核心设计哲学是 "Real-time Feedback, Intuitive Control" (实时反馈，直观控制)：
- 实时通信：SSE/WebSocket 推送执行状态
- 可视化：ECharts 图表展示测试数据
- 代码编辑：Monaco Editor 在线编写测试脚本
- 响应式：适配桌面和平板设备

1.2 技术选型

4. 实时通信设计

4.1 SSE (Server-Sent Events) 客户端

// services/sse.ts

class SSEClient {
  private eventSource: EventSource | null = null;
  private listeners: Map<string, Set<Function>> = new Map();
  
  connect(url: string): void {
    this.eventSource = new EventSource(url);
    
    this.eventSource.onmessage = (event) => {
      const data = JSON.parse(event.data);
      this.emit(data.type, data.payload);
    };
    
    this.eventSource.onerror = (error) => {
      console.error('SSE connection error:', error);
      this.reconnect(url);
    };
  }
  
  on(eventType: string, callback: Function): void {
    if (!this.listeners.has(eventType)) {
      this.listeners.set(eventType, new Set());
    }
    this.listeners.get(eventType)!.add(callback);
  }
  
  off(eventType: string, callback: Function): void {
    this.listeners.get(eventType)?.delete(callback);
  }
  
  private emit(eventType: string, payload: any): void {
    this.listeners.get(eventType)?.forEach(cb => cb(payload));
  }
  
  disconnect(): void {
    this.eventSource?.close();
    this.eventSource = null;
  }
  
  private reconnect(url: string): void {
    setTimeout(() => this.connect(url), 3000);
  }
}

export const sseClient = new SSEClient();

4.2 SSE 事件类型

| 事件类型 | 说明 | Payload |
|---------|------|---------|
| execution.started | 执行开始 | { execution_id, tc_id, started_at } |
| execution.step | 步骤执行 | { step_index, status, screenshot_url } |
| execution.completed | 执行完成 | { execution_id, result, duration } |
| execution.failed | 执行失败 | { execution_id, error, failed_step } |
| healing.started | 自愈开始 | { execution_id, failed_step } |
| healing.completed | 自愈完成 | { execution_id, healed } |

4.3 使用示例

// hooks/useSSE.ts

import { useEffect } from 'react';
import { sseClient } from '@/services/sse';
import { useExecutionStore } from '@/stores/useExecutionStore';

export function useExecutionSSE(executionId: string) {
  const { updateExecution, addStep } = useExecutionStore();
  
  useEffect(() => {
    sseClient.connect(`/api/v1/executions/${executionId}/stream`);
    
    const handleStep = (payload: StepPayload) => {
      addStep(executionId, payload);
    };
    
    const handleCompleted = (payload: CompletedPayload) => {
      updateExecution(executionId, { status: 'completed', ...payload });
    };
    
    sseClient.on('execution.step', handleStep);
    sseClient.on('execution.completed', handleCompleted);
    
    return () => {
      sseClient.off('execution.step', handleStep);
      sseClient.off('execution.completed', handleCompleted);
      sseClient.disconnect();
    };
  }, [executionId]);
}

5. 状态管理设计

5.1 Zustand Store 示例

// stores/useExecutionStore.ts

import { create } from 'zustand';

interface Execution {
  id: string;
  tcId: string;
  status: 'pending' | 'running' | 'completed' | 'failed';
  steps: Step[];
  startedAt: Date;
  completedAt?: Date;
}

interface ExecutionStore {
  executions: Map<string, Execution>;
  currentExecutionId: string | null;
  
  // Actions
  startExecution: (tcId: string) => Promise<string>;
  updateExecution: (id: string, updates: Partial<Execution>) => void;
  addStep: (executionId: string, step: Step) => void;
  setCurrentExecution: (id: string) => void;
}

export const useExecutionStore = create<ExecutionStore>((set, get) => ({
  executions: new Map(),
  currentExecutionId: null,
  
  startExecution: async (tcId: string) => {
    const response = await api.post('/executions', { tc_id: tcId });
    const execution: Execution = {
      id: response.data.execution_id,
      tcId,
      status: 'pending',
      steps: [],
      startedAt: new Date()
    };
    
    set(state => ({
      executions: new Map(state.executions).set(execution.id, execution),
      currentExecutionId: execution.id
    }));
    
    return execution.id;
  },
  
  updateExecution: (id, updates) => {
    set(state => {
      const executions = new Map(state.executions);
      const existing = executions.get(id);
      if (existing) {
        executions.set(id, { ...existing, ...updates });
      }
      return { executions };
    });
  },
  
  addStep: (executionId, step) => {
    set(state => {
      const executions = new Map(state.executions);
      const existing = executions.get(executionId);
      if (existing) {
        executions.set(executionId, {
          ...existing,
          steps: [...existing.steps, step]
        });
      }
      return { executions };
    });
  },
  
  setCurrentExecution: (id) => set({ currentExecutionId: id })
}));

Button 组件：

interface ButtonProps {
  variant: 'primary' | 'secondary' | 'danger' | 'ghost';
  size: 'sm' | 'md' | 'lg';
  loading?: boolean;
  disabled?: boolean;
  icon?: React.ReactNode;
  children: React.ReactNode;
  onClick?: () => void;
}

export function Button({ variant, size, loading, ...props }: ButtonProps) {
  const variants = {
    primary: 'bg-blue-500 hover:bg-blue-600 text-white',
    secondary: 'bg-gray-100 hover:bg-gray-200 text-gray-900',
    danger: 'bg-red-500 hover:bg-red-600 text-white',
    ghost: 'hover:bg-gray-100 text-gray-700'
  };
  
  const sizes = {
    sm: 'px-3 py-1.5 text-sm',
    md: 'px-4 py-2 text-base',
    lg: 'px-6 py-3 text-lg'
  };
  
  return (
    <button
      className={`
        ${variants[variant]}
        ${sizes[size]}
        rounded-lg font-medium
        flex items-center gap-2
        transition-colors
        disabled:opacity-50 disabled:cursor-not-allowed
      `}
      disabled={loading || props.disabled}
      {...props}
    >
      {loading && <Spinner size="sm" />}
      {props.icon}
      {props.children}
    </button>
  );
}

import Editor from '@monaco-editor/react';

interface TCIREditorProps {
  value: string;
  onChange: (value: string) => void;
  readOnly?: boolean;
}

export function TCIREditor({ value, onChange, readOnly }: TCIREditorProps) {
  const handleEditorMount = (editor: any, monaco: any) => {
    // 注册 TC-IR JSON Schema
    monaco.languages.json.jsonDefaults.setDiagnosticsOptions({
      validate: true,
      schemas: [{
        uri: 'tc-ir-schema',
        fileMatch: ['*'],
        schema: TC_IR_SCHEMA
      }]
    });
  };
  
  return (
    <Editor
      height="400px"
      language="json"
      theme="vs-dark"
      value={value}
      onChange={(val) => onChange(val || '')}
      onMount={handleEditorMount}
      options={{
        readOnly,
        minimap: { enabled: false },
        fontSize: 14,
        lineNumbers: 'on',
        wordWrap: 'on'
      }}
    />
  );
}

// services/api.ts

import axios from 'axios';

const api = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL || '/api/v1',
  timeout: 30000,
  headers: {
    'Content-Type': 'application/json'
  }

// 请求拦截器
api.interceptors.request.use(config => {
  const token = localStorage.getItem('auth_token');
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

// 响应拦截器
api.interceptors.response.use(
  response => response,
  error => {
    if (error.response?.status === 401) {
      // 跳转登录
      window.location.href = '/login';
    }
    return Promise.reject(error);
  }
);

// API 方法封装
