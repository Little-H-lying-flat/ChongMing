1. 系统概述 (System Overview)

1.1 定义与愿景

凤凰涅槃层 (Phoenix Nirvana Layer) 是重明架构中负责 "固" (Solidify) 和 "生" (Generate) 的核心组件。它是"三态生命周期"中液态→固态转化的关键引擎，将 AI 探索产生的执行轨迹（Trace Log）编译为可维护、可版本控制的测试脚本资产。

其核心设计哲学是 "Trace-to-Code, Intent-Preserved" (轨迹代码化，意图保真)：
- 轨迹提取：从液态执行中提取关键操作序列
- 意图保真：保留原始测试意图和业务语义
- 代码生成：编译为符合规范的 Python/Pytest 脚本
- 资产管理：支持版本控制、参数化、数据驱动

1.2 核心职责
- 轨迹解析 (Trace Parsing)：解析 AUI-IR/API-IR 执行轨迹
- 代码编译 (Code Compilation)：生成 Python 测试脚本
- 意图注释 (Intent Annotation)：保留业务语义注释
- 参数提取 (Parameter Extraction)：识别可参数化的数据
- 资产版本 (Asset Versioning)：管理脚本版本历史

3.1 输入格式

|---------|------|------|

3.2 轨迹分析器

class TraceParser:
    def parse(self, trace_log: TraceLog) -> ParsedTrace:
        steps = []
        
        for entry in trace_log.entries:
            if entry.type == "UI_ACTION":
                step = self._parse_ui_action(entry)
            elif entry.type == "API_CALL":
                step = self._parse_api_call(entry)
            elif entry.type == "ASSERTION":
                step = self._parse_assertion(entry)
            elif entry.type == "WAIT":
                step = self._parse_wait(entry)
            else:
                continue
            
            steps.append(step)
        
        return ParsedTrace(
            tc_id=trace_log.tc_id,
            tc_name=trace_log.tc_name,
            steps=steps,
            metadata=trace_log.metadata
        )
    
    def _parse_ui_action(self, entry: TraceEntry) -> UIStep:
        return UIStep(
            action=entry.action,  # click, input, hover, etc.
            strategy=entry.strategy,  # visual or dom
            target=entry.target,
            coordinates=entry.coordinates,
            selector=entry.selector,
            value=entry.value,
            screenshot_before=entry.screenshot_before,
            screenshot_after=entry.screenshot_after
        )

3.3 抽象语法树构建

class ASTBuilder:
    def build(self, parsed_trace: ParsedTrace) -> TestAST:
        ast = TestAST(
            name=self._generate_test_name(parsed_trace.tc_name),
            docstring=parsed_trace.tc_name,
            fixtures=["page"],  # Playwright fixture
            body=[]
        )
        
        for step in parsed_trace.steps:
            node = self._step_to_node(step)
            ast.body.append(node)
        
        return ast
    
    def _step_to_node(self, step: Step) -> ASTNode:
        if isinstance(step, UIStep):
            return UIActionNode(
                action=step.action,
                locator=self._build_locator(step),
                value=step.value,
                comment=step.description
            )
        elif isinstance(step, APIStep):
            return APICallNode(
                method=step.method,
                url=step.url,
                headers=step.headers,
                body=step.body,
                assertions=step.assertions
            )

4. 优化层 (Optimizer Layer)

4.1 去重优化器

去除冗余步骤，合并相似操作：

class DedupOptimizer:
    def optimize(self, ast: TestAST) -> TestAST:
        optimized_body = []
        prev_node = None
        
        for node in ast.body:
            # 合并连续的相同定位器等待
            if self._is_redundant_wait(prev_node, node):
                continue
            
            # 合并连续输入到同一元素
            if self._is_consecutive_input(prev_node, node):
                prev_node.value += node.value
                continue
            
            optimized_body.append(node)
            prev_node = node
        
        ast.body = optimized_body
        return ast

4.2 参数提取器

识别可参数化的数据，支持数据驱动：

class ParameterExtractor:
    # 可参数化的数据模式
    PATTERNS = {
        "username": r"user\d+|admin|test",
        "password": r".*password.*|pass\d+",
        "email": r"[\w.]+@[\w.]+",
        "phone": r"1[3-9]\d{9}",
        "date": r"\d{4}-\d{2}-\d{2}",
        "amount": r"\d+\.?\d*"
    }
    
    def extract(self, ast: TestAST) -> tuple[TestAST, list[Parameter]]:
        parameters = []
        
        for node in ast.body:
            if hasattr(node, 'value') and node.value:
                param = self._detect_parameter(node.value)
                if param:
                    # 替换为参数引用
                    node.value = f"{{{{ {param.name} }}}}"
                    parameters.append(param)
        
        return ast, parameters
    
    def _detect_parameter(self, value: str) -> Parameter | None:
        for param_type, pattern in self.PATTERNS.items():
            if re.match(pattern, value, re.IGNORECASE):
                return Parameter(
                    name=f"test_{param_type}",
                    type=param_type,
                    default_value=value
                )
        return None

class AssertionEnhancer:
    def enhance(self, ast: TestAST, trace: ParsedTrace) -> TestAST:
        enhanced_body = []
        
        for i, node in enumerate(ast.body):
            enhanced_body.append(node)
            
            # 导航后添加 URL 断言

                    selector=".success-message, .toast-success",
                    should_exist=True
                ))
        
        ast.body = enhanced_body
        return ast

5. 编译层 (Compiler Layer)

5.1 代码生成器

class PythonCompiler:
    def compile(self, ast: TestAST, parameters: list[Parameter]) -> str:
        template = self._load_template("pytest_template.j2")
        
        code = template.render(
            test_name=ast.name,
            docstring=ast.docstring,
            fixtures=ast.fixtures,
            parameters=parameters,
            steps=self._compile_steps(ast.body),
            imports=self._generate_imports(ast)
        )
        
        return code
    
    def _compile_steps(self, nodes: list[ASTNode]) -> list[str]:
        code_lines = []
        
        for node in nodes:
            if isinstance(node, UIActionNode):
                code_lines.append(self._compile_ui_action(node))
            elif isinstance(node, APICallNode):
                code_lines.append(self._compile_api_call(node))
            elif isinstance(node, AssertNode):
                code_lines.append(self._compile_assertion(node))
        
        return code_lines

5.3 UI 操作编译规则

| AUI-IR 动作 | Playwright 代码 |
|------------|-----------------|
| click (visual) | `page.mouse.click(x, y)` |
| click (dom) | `page.locator("selector").click()` |
| input | `page.locator("selector").fill("value")` |
| hover | `page.locator("selector").hover()` |
| scroll | `page.mouse.wheel(0, delta)` |
| navigate | `page.goto("url")` |
| wait | `page.wait_for_selector("selector")` |

def _compile_ui_action(self, node: UIActionNode) -> str:
    if node.action == "click":
        if node.strategy == "visual":
            return f'page.mouse.click({node.coordinates[0]}, {node.coordinates[1]})'
        else:
            return f'page.locator("{node.selector}").click()'
    
    elif node.action == "input":
        return f'page.locator("{node.selector}").fill("{node.value}")'
    
    elif node.action == "navigate":
        return f'page.goto("{node.target}")'
    
    elif node.action == "hover":
        return f'page.locator("{node.selector}").hover()'

5.4 API 调用编译规则

def _compile_api_call(self, node: APICallNode) -> str:
    code_lines = []
    
    # 构建请求
    code_lines.append(f'response = client.{node.method.lower()}(')
    code_lines.append(f'    "{node.url}",')
    
    if node.headers:
        code_lines.append(f'    headers={repr(node.headers)},')
    
    if node.body:
        code_lines.append(f'    json={repr(node.body)},')
    
    code_lines.append(')')
    
    # 添加断言
    for assertion in node.assertions:
        if assertion.type == "status_code":
            code_lines.append(f'assert response.status_code == {assertion.expected}')
        elif assertion.type == "json_path":
            code_lines.append(
                f'assert response.json(){assertion.path} == {repr(assertion.expected)}'
            )
    
    return '\n'.join(code_lines)

6.1 意图注释器

保留业务语义，让代码更易维护：

class IntentAnnotator:
    def annotate(self, code: str, trace: ParsedTrace) -> str:
        # 1. 添加文件头注释
        header = self._generate_header(trace)
        
        # 2. 为每个步骤添加业务注释
        annotated_code = code
        for step in trace.steps:
            if step.description:
                # 在代码行前插入注释
                annotated_code = self._insert_comment(
                    annotated_code,
                    step.code_line,
                    step.description
                )
        
        # 3. 添加来源追溯
        footer = self._generate_footer(trace)
        
        return header + annotated_code + footer

class AssetVersioner:
    def __init__(self, repo_path: str):
        self.repo = git.Repo(repo_path)
    
    def save(self, test_file: TestFile) -> str:
        # 1. 写入文件
        file_path = self._get_file_path(test_file)
        with open(file_path, 'w') as f:
            f.write(test_file.content)
        
        # 2. 检测变更
        if self._has_changes(file_path):
            # 3. 生成变更摘要
            diff = self._get_diff(file_path)
            commit_message = self._generate_commit_message(test_file, diff)
            
            # 4. 提交
            self.repo.index.add([file_path])
            commit = self.repo.index.commit(commit_message)
            
            return commit.hexsha
        
        return None
    
    def _generate_commit_message(self, test_file: TestFile, diff: str) -> str:
        return f"""[Phoenix-Auto] Update {test_file.name}

Source: {test_file.source_tc_id}
Changes:
{self._summarize_diff(diff)}

Generated by Phoenix Nirvana Layer v1.0
"""

def get_version_history(self, file_path: str) -> list[VersionInfo]:
    commits = list(self.repo.iter_commits(paths=file_path, max_count=20))
    
    return [
        VersionInfo(
            version=commit.hexsha[:8],
            author=commit.author.name,
            date=commit.committed_datetime,
            message=commit.message,

    ]

8. 编译策略 (Compilation Strategies)

8.1 策略选择

| 策略 | 描述 | 适用场景 |
|------|------|---------|
| 精确复现 | 保留所有步骤，包括等待和中间状态 | 调试、回归测试 |
| 简化优化 | 去除冗余，合并步骤 | 持续集成 |
| 数据驱动 | 提取参数，支持多数据集 | 批量测试 |
| 混合模式 | 视觉 + DOM 双保险 | 稳定性要求高 |

9. 自愈脚本编译 (Self-Healing Compilation)

9.1 气态→固态编译

当脚本自愈成功后，将修复策略编译回脚本：

class SelfHealingCompiler:
    def compile_healing(self, original_script: str, 
                        healing_record: HealingRecord) -> str:
        # 1. 解析原脚本
        ast = parse(original_script)
        
        # 2. 定位失败步骤
        failed_node = self._find_node(ast, healing_record.failed_step)
        
        # 3. 应用修复
        if healing_record.healing_type == "locator_update":
            # 更新定位器
            failed_node.selector = healing_record.new_selector
        
        elif healing_record.healing_type == "strategy_switch":
            # 切换策略 (visual → dom 或 dom → visual)
            failed_node.strategy = healing_record.new_strategy
            failed_node.coordinates = healing_record.new_coordinates
        
        # 4. 添加自愈注释
        self._add_healing_comment(failed_node, healing_record)
        
        # 5. 重新生成代码
        return unparse(ast)

12.2 脚本版本管理 (Script Version Management)

12.2.1 Git 管理器实现

class GitManager:
    """脚本版本管理器 - 自动提交生成的测试脚本到 Git 仓库"""
    
    def __init__(self, config: dict):
        self.repo_path = Path(config.get("repo_path", "./test_assets"))
        self.repo = git.Repo(self.repo_path)
        self.branch_prefix = config.get("branch_prefix", "phoenix-auto")
        self.auto_commit = config.get("auto_commit", True)
        self.auto_pr = config.get("auto_pr", False)
    
    async def commit_script(self, script: CompiledScript) -> GitCommitResult:
        """提交编译后的脚本"""
        
        # 1. 切换/创建分支
        branch_name = f"{self.branch_prefix}/{script.tc_id}"
        self._ensure_branch(branch_name)
        
        # 2. 写入脚本文件
        script_path = self.repo_path / script.relative_path
        script_path.parent.mkdir(parents=True, exist_ok=True)
        script_path.write_text(script.content, encoding="utf-8")
        
        # 3. Git add
        self.repo.index.add([str(script_path)])
        
        # 4. 生成提交信息
        commit_msg = self._generate_commit_message(script)
        
        # 5. Git commit
        commit = self.repo.index.commit(commit_msg)
        
        return GitCommitResult(
            commit_sha=commit.hexsha,
            branch=branch_name,
            file_path=str(script_path),
            message=commit_msg
        )
    
    def _generate_commit_message(self, script: CompiledScript) -> str:
        """生成标准化提交信息"""
        return f"""[Phoenix] {script.tc_id}: {script.name}

自动编译自 Trace ID: {script.source_trace_id}
编译策略: {script.compilation_strategy}
生成步骤数: {script.step_count}
参数化字段: {', '.join(script.parameters)}

Co-authored-by: Phoenix Nirvana Layer <phoenix@chongming.ai>
"""
    
    def _ensure_branch(self, branch_name: str):
        """确保分支存在并切换"""
        if branch_name not in [b.name for b in self.repo.branches]:
            self.repo.create_head(branch_name)
        self.repo.heads[branch_name].checkout()

12.2.4 版本回滚

class ScriptVersionManager:
    """脚本版本历史管理"""
    
    async def get_history(self, script_path: str, limit: int = 10) -> list:
        """获取脚本的版本历史"""
        commits = list(self.repo.iter_commits(paths=script_path, max_count=limit))
        return [
            {
                "sha": c.hexsha[:7],
                "message": c.message.split("\n")[0],
                "author": str(c.author),
                "date": c.committed_datetime.isoformat(),
                "diff_stats": c.stats.total
            }
            for c in commits
        ]
    
    async def rollback(self, script_path: str, target_sha: str) -> bool:
        """回滚脚本到指定版本"""
        # 1. 获取目标版本的内容
        blob = self.repo.commit(target_sha).tree / script_path
        content = blob.data_stream.read().decode("utf-8")
        
        # 2. 写入当前文件
        Path(self.repo_path / script_path).write_text(content)
        
        # 3. 提交回滚
        self.repo.index.add([script_path])
        self.repo.index.commit(f"[Phoenix] Rollback {script_path} to {target_sha[:7]}")
        
        return True
