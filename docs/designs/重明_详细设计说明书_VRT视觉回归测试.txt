1. 系统概述 (System Overview)

1.1 定义与愿景

VRT 视觉回归测试模块 (Visual Regression Testing Module) 是重明架构中负责 "视觉快照对比" 的质量保障组件。它像一个精确的视觉对比仪，自动捕获页面截图并与基线进行像素级、结构级对比，确保 UI 变更不会引入意外的视觉缺陷。

其核心设计哲学是 "Pixel-Perfect, Change-Aware" (像素精准，变化感知)：
- 多策略对比：支持像素级、结构级、语义级三种对比策略
- 智能差异识别：利用 AI 区分"预期变更"与"意外缺陷"
- 基线版本管理：Git 集成的可追溯基线存储
- 跨浏览器验证：Playwright 多浏览器截图能力

1.2 核心价值

1. **UI 一致性保障** - 自动检测页面布局、颜色、字体变化
2. **回归测试自动化** - 每次部署自动触发视觉对比
3. **设计稿还原度检查** - 对比实际渲染与设计稿差异
4. **跨浏览器兼容性** - 检测不同浏览器的渲染差异

```python
from dataclasses import dataclass
from enum import Enum
from typing import Optional, List
from datetime import datetime

class ComparisonStrategy(Enum):
    PIXEL = "pixel"           # 像素级对比
    STRUCTURAL = "structural"  # 结构级对比 (DOM + 布局)
    SSIM = "ssim"             # 结构相似性指数
    AI_SEMANTIC = "ai"        # AI 语义对比

class DiffCategory(Enum):
    EXPECTED_CHANGE = "expected"    # 预期变更
    UNEXPECTED_BUG = "bug"          # 意外缺陷
    LAYOUT_SHIFT = "layout"         # 布局偏移
    COLOR_CHANGE = "color"          # 颜色变化
    CONTENT_CHANGE = "content"      # 内容变化
    FONT_CHANGE = "font"            # 字体变化
    ANIMATION_DIFF = "animation"    # 动画差异

@dataclass
class VisualBaseline:
    """视觉基线"""
    baseline_id: str
    page_url: str
    page_name: str
    screenshot_path: str
    browser: str              # chromium, firefox, webkit
    viewport: dict            # {width, height}
    created_at: datetime
    approved_by: Optional[str]
    git_commit: str
    metadata: dict

@dataclass
class VisualDiff:
    """视觉差异"""
    diff_id: str
    baseline_id: str
    current_screenshot: str
    diff_image_path: str
    diff_percentage: float    # 差异百分比
    diff_pixels: int          # 差异像素数
    ssim_score: float         # 结构相似度 (0-1)
    diff_regions: List[dict]  # 差异区域坐标
    category: DiffCategory
    confidence: float
    
@dataclass
class VRTReport:
    """VRT 测试报告"""
```

```python
from playwright.async_api import Page, Browser
import asyncio
from pathlib import Path

class ScreenshotCapturer:
    """多浏览器截图捕获器"""
    
    def __init__(self, config: dict):
        self.output_dir = Path(config.get("output_dir", "./screenshots"))
        self.browsers = config.get("browsers", ["chromium"])
        self.viewports = config.get("viewports", [
            {"width": 1920, "height": 1080, "name": "desktop"},
            {"width": 768, "height": 1024, "name": "tablet"},
            {"width": 375, "height": 812, "name": "mobile"}
        ])
    
    async def capture_page(
        self, 
        page: Page, 
        page_name: str,
        full_page: bool = True,
        wait_for_stability: bool = True
    ) -> dict:
        """捕获页面截图"""
        
        # 1. 等待页面稳定 (集成智能等待)
        if wait_for_stability:
            await self._wait_for_visual_stability(page)
        
        # 2. 隐藏动态内容 (防止动画干扰)
        await self._mask_dynamic_content(page)
        
        # 3. 多视口截图
        screenshots = {}
        for viewport in self.viewports:
            await page.set_viewport_size(viewport)
            await asyncio.sleep(0.3)  # 等待重绘
            
            screenshot_path = self.output_dir / f"{page_name}_{viewport['name']}.png"
            await page.screenshot(
                path=str(screenshot_path),
                full_page=full_page,
                animations="disabled"
            )
            screenshots[viewport['name']] = str(screenshot_path)
        
        return screenshots
    
    async def _wait_for_visual_stability(self, page: Page) -> None:
        """等待视觉稳定 (复用智能等待机制)"""
        # 使用 SmartWait 的视觉稳定检测
        await page.evaluate("""
            () => new Promise(resolve => {
                let lastHash = '';
                const checkStability = () => {
                    const canvas = document.createElement('canvas');
                    // 简化的视觉哈希检测
                    setTimeout(resolve, 500);
                };
                checkStability();
            })
        """)
    
    async def _mask_dynamic_content(self, page: Page) -> None:
        """遮盖动态内容 (时间戳、广告、动画)"""
        await page.evaluate("""
            () => {
                // 隐藏常见动态元素
                const selectors = [
                    '[data-testid="timestamp"]',
                    '[class*="ad"]',
                    '[class*="carousel"]',
                    'video', 'iframe[src*="youtube"]'
                ];
                selectors.forEach(sel => {
                    document.querySelectorAll(sel).forEach(el => {
                        el.style.visibility = 'hidden';
                    });
                });
            }
        """)
```

4.2 图像对比引擎

```python
from PIL import Image
import numpy as np
from skimage.metrics import structural_similarity as ssim
from dataclasses import dataclass
import cv2

class ComparisonEngine:
    """多策略图像对比引擎"""
    
    def __init__(self, config: dict):
        self.threshold = config.get("threshold", 0.01)  # 1% 容差
        self.ssim_threshold = config.get("ssim_threshold", 0.95)
    
    def compare(
        self, 
        baseline_path: str, 
        current_path: str,
        strategy: ComparisonStrategy = ComparisonStrategy.PIXEL
    ) -> VisualDiff:
        """执行图像对比"""
        
        baseline_img = Image.open(baseline_path)
        current_img = Image.open(current_path)
        
        # 尺寸检查
        if baseline_img.size != current_img.size:
            return self._handle_size_mismatch(baseline_img, current_img)
        
        if strategy == ComparisonStrategy.PIXEL:
            return self._pixel_compare(baseline_img, current_img)
        elif strategy == ComparisonStrategy.SSIM:
            return self._ssim_compare(baseline_img, current_img)
        elif strategy == ComparisonStrategy.STRUCTURAL:
            return self._structural_compare(baseline_img, current_img)
        elif strategy == ComparisonStrategy.AI_SEMANTIC:
            return self._ai_semantic_compare(baseline_img, current_img)
    
    def _pixel_compare(self, baseline: Image, current: Image) -> VisualDiff:
        """像素级对比"""
        baseline_arr = np.array(baseline)
        current_arr = np.array(current)
        
        # 计算差异
        diff = np.abs(baseline_arr.astype(int) - current_arr.astype(int))
        diff_mask = np.any(diff > 5, axis=2)  # RGB 任一通道差异 > 5
        
        diff_pixels = np.sum(diff_mask)
        total_pixels = diff_mask.size
        diff_percentage = (diff_pixels / total_pixels) * 100
        
        # 生成差异图
        diff_image = self._generate_diff_image(baseline_arr, current_arr, diff_mask)
        
        # 提取差异区域
        diff_regions = self._extract_diff_regions(diff_mask)
        
        return VisualDiff(
            diff_id=f"diff_{hash(str(baseline_arr))}",
            baseline_id="",
            current_screenshot=str(current),
            diff_image_path=diff_image,
            diff_percentage=diff_percentage,
            diff_pixels=int(diff_pixels),
            ssim_score=0.0,
            diff_regions=diff_regions,
            category=DiffCategory.UNEXPECTED_BUG if diff_percentage > self.threshold else DiffCategory.EXPECTED_CHANGE,
            confidence=0.9
        )
    
    def _ssim_compare(self, baseline: Image, current: Image) -> VisualDiff:
        """SSIM 结构相似性对比"""
        baseline_gray = np.array(baseline.convert('L'))
        current_gray = np.array(current.convert('L'))
        
        score, diff = ssim(baseline_gray, current_gray, full=True)
        
        return VisualDiff(
            diff_id=f"ssim_{hash(str(baseline_gray))}",
            baseline_id="",
            current_screenshot="",
            diff_image_path="",
            diff_percentage=(1 - score) * 100,
            diff_pixels=0,
            ssim_score=score,
            diff_regions=[],
            category=DiffCategory.UNEXPECTED_BUG if score < self.ssim_threshold else DiffCategory.EXPECTED_CHANGE,
            confidence=score
        )
    
    def _generate_diff_image(
        self, 
        baseline: np.ndarray, 
        current: np.ndarray, 
        diff_mask: np.ndarray
    ) -> str:
        """生成差异可视化图"""
        # 创建差异叠加图
        overlay = current.copy()
        overlay[diff_mask] = [255, 0, 255]  # 粉红色高亮差异区域
        
        # 混合原图与差异
        blended = cv2.addWeighted(current, 0.7, overlay, 0.3, 0)
        
        # 保存差异图
        diff_path = f"/tmp/diff_{hash(str(baseline))}.png"
        cv2.imwrite(diff_path, cv2.cvtColor(blended, cv2.COLOR_RGB2BGR))
        
        return diff_path
    
    def _extract_diff_regions(self, diff_mask: np.ndarray) -> List[dict]:
        """提取差异区域边界框"""
        # 使用连通组件分析
        labeled, num_features = cv2.connectedComponents(diff_mask.astype(np.uint8))
        
        regions = []
        for i in range(1, num_features + 1):
            ys, xs = np.where(labeled == i)
            if len(xs) > 0:
                regions.append({
                    "x": int(xs.min()),
                    "y": int(ys.min()),
                    "width": int(xs.max() - xs.min()),
                    "height": int(ys.max() - ys.min()),
                    "pixels": len(xs)
                })
        
        return regions
```

4.3 基线管理器

```python
import git
from pathlib import Path
import json
import hashlib

class BaselineManager:
    """Git 集成的基线版本管理器"""
    
    def __init__(self, config: dict):
        self.baseline_dir = Path(config.get("baseline_dir", "./vrt_baselines"))
        self.repo = git.Repo(config.get("git_repo", "."))
    
    def get_baseline(self, page_name: str, viewport: str) -> Optional[VisualBaseline]:
        """获取页面基线"""
        baseline_path = self.baseline_dir / f"{page_name}_{viewport}.png"
        meta_path = self.baseline_dir / f"{page_name}_{viewport}.json"
        
        if not baseline_path.exists():
            return None
        
        metadata = json.loads(meta_path.read_text()) if meta_path.exists() else {}
        
        return VisualBaseline(
            baseline_id=hashlib.md5(baseline_path.read_bytes()).hexdigest(),
            page_url=metadata.get("url", ""),
            page_name=page_name,
            screenshot_path=str(baseline_path),
            browser=metadata.get("browser", "chromium"),
            viewport={"name": viewport},
            created_at=metadata.get("created_at"),
            approved_by=metadata.get("approved_by"),
            git_commit=metadata.get("git_commit", ""),
            metadata=metadata
        )
    
    def update_baseline(
        self, 
        page_name: str, 
        viewport: str, 
        screenshot_path: str,
        approver: str
    ) -> VisualBaseline:
        """更新基线 (需要审批)"""
        
        # 1. 复制截图到基线目录
        dest_path = self.baseline_dir / f"{page_name}_{viewport}.png"
        Path(screenshot_path).replace(dest_path)
        
        # 2. 记录元数据
        metadata = {
            "url": "",
            "browser": "chromium",
            "created_at": datetime.now().isoformat(),
            "approved_by": approver,
            "git_commit": self.repo.head.commit.hexsha
        }
        meta_path = self.baseline_dir / f"{page_name}_{viewport}.json"
        meta_path.write_text(json.dumps(metadata, indent=2))
        
        # 3. Git 提交
        self.repo.index.add([str(dest_path), str(meta_path)])
        self.repo.index.commit(f"[VRT] Update baseline: {page_name}_{viewport}")
        
        return self.get_baseline(page_name, viewport)
    
    def list_baselines(self) -> List[VisualBaseline]:
        """列出所有基线"""
        baselines = []
        for png_file in self.baseline_dir.glob("*.png"):
            name_parts = png_file.stem.rsplit("_", 1)
            if len(name_parts) == 2:
                page_name, viewport = name_parts
                baseline = self.get_baseline(page_name, viewport)
                if baseline:
                    baselines.append(baseline)
        return baselines
```

```python
class AIDiffClassifier:
    """AI 驱动的差异智能分类器"""
    
    def __init__(self, llm_client):
        self.llm = llm_client
    
    async def classify_diff(
        self, 
        baseline_path: str, 
        current_path: str,
        diff_regions: List[dict],
        context: dict
    ) -> DiffCategory:
        """使用 AI 分类差异类型"""
        
        prompt = f"""
分析以下视觉差异，判断类型：

差异区域：
{json.dumps(diff_regions, indent=2)}

上下文信息：
- 页面: {context.get('page_name')}
- 最近代码变更: {context.get('recent_commits', [])}
- 相关 PR: {context.get('related_pr', 'N/A')}

请分类为以下类型之一：
1. expected - 预期变更 (如：已知的 UI 更新)
2. bug - 意外缺陷 (如：布局错乱、元素消失)
3. layout - 布局偏移
4. color - 颜色变化
5. content - 内容变化
6. font - 字体变化

返回 JSON 格式：
{{"category": "xxx", "confidence": 0.95, "reason": "..."}}
"""
        
        response = await self.llm.generate(prompt, model="qwen3-max")
        result = json.loads(response)
        
        return DiffCategory(result["category"])
    
    async def suggest_action(self, diff: VisualDiff) -> dict:
        """建议处理动作"""
        if diff.category == DiffCategory.EXPECTED_CHANGE:
            return {
                "action": "approve_baseline",
                "message": "建议更新基线为当前截图"
            }
        elif diff.category == DiffCategory.UNEXPECTED_BUG:
            return {
                "action": "create_issue",
                "message": "检测到视觉缺陷，建议创建 Issue",
                "priority": "high" if diff.diff_percentage > 5 else "medium"
            }
        else:
            return {
                "action": "manual_review",
                "message": "需要人工确认差异原因"
            }
```

```python
                <h4>当前</h4>

            </div>

                <img src="{{ result.diff_path }}" alt="diff">
            </div>
        </div>

</html>

        Path(report_path).write_text(html_content)
        
        return report_path
```

```python
def test_login_page_visual(page: Page):
```
