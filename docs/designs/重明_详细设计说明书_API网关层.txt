1. 系统概述 (System Overview)

1.1 定义与愿景

API 网关层 (API Gateway Layer) 是重明架构中负责 "连" (Connect) 和 "控" (Control) 的核心组件。它是所有外部请求进入系统的唯一入口，承载着路由分发、认证鉴权、限流熔断等关键网关职能。

其核心设计哲学是 "Single Entry, Unified Control" (单一入口，统一管控)：
- 统一入口：所有 API 请求通过网关路由
- 认证鉴权：JWT Token 验证与权限控制
- 流量控制：限流、熔断、降级
- 可观测性：请求日志、指标采集、链路追踪

1.2 技术选型

3. 核心中间件设计

3.1 认证中间件

# middleware/auth.py

from fastapi import Request, HTTPException
from fastapi.security import HTTPBearer
import jwt

class AuthMiddleware:
    def __init__(self, secret_key: str, algorithm: str = "HS256"):
        self.secret_key = secret_key
        self.algorithm = algorithm
        self.public_paths = ["/api/v1/auth/login", "/api/v1/auth/register", "/health"]
    
    async def __call__(self, request: Request, call_next):
        # 公开路径跳过认证
        if request.url.path in self.public_paths:
            return await call_next(request)
        
        # 提取 Token
        auth_header = request.headers.get("Authorization")
        if not auth_header or not auth_header.startswith("Bearer "):
            raise HTTPException(status_code=401, detail="Missing authentication token")
        
        token = auth_header.split(" ")[1]
        
        try:
            # 验证 Token
            payload = jwt.decode(token, self.secret_key, algorithms=[self.algorithm])
            request.state.user_id = payload.get("sub")
            request.state.user_role = payload.get("role")
        except jwt.ExpiredSignatureError:
            raise HTTPException(status_code=401, detail="Token has expired")
        except jwt.InvalidTokenError:
            raise HTTPException(status_code=401, detail="Invalid token")
        
        return await call_next(request)

3.2 限流中间件

# middleware/rate_limit.py

from fastapi import Request, HTTPException
import redis.asyncio as redis
import time

class RateLimitMiddleware:
    def __init__(self, redis_client: redis.Redis, rate: int = 100, window: int = 60):
        """
        :param rate: 时间窗口内允许的最大请求数
        :param window: 时间窗口 (秒)
        """
        self.redis = redis_client
        self.rate = rate
        self.window = window
    
    async def __call__(self, request: Request, call_next):
        # 以用户 ID 或 IP 作为限流 Key
        user_id = getattr(request.state, 'user_id', None)
        key = f"rate_limit:{user_id or request.client.host}"
        
        # 滑动窗口计数
        current_time = int(time.time())
        window_start = current_time - self.window
        
        async with self.redis.pipeline() as pipe:
            # 移除过期请求
            await pipe.zremrangebyscore(key, 0, window_start)
            # 添加当前请求
            await pipe.zadd(key, {str(current_time): current_time})
            # 统计窗口内请求数
            await pipe.zcard(key)
            # 设置过期时间
            await pipe.expire(key, self.window)
            results = await pipe.execute()
        
        request_count = results[2]
        
        if request_count > self.rate:
            raise HTTPException(
                status_code=429,
                detail=f"Rate limit exceeded. Max {self.rate} requests per {self.window}s"
            )
        
        response = await call_next(request)
        
        # 添加限流头
        response.headers["X-RateLimit-Limit"] = str(self.rate)
        response.headers["X-RateLimit-Remaining"] = str(max(0, self.rate - request_count))
        
        return response

3.3 日志中间件

# middleware/logging.py

from fastapi import Request
import logging
import time
import uuid

logger = logging.getLogger("api_gateway")

class LoggingMiddleware:
    async def __call__(self, request: Request, call_next):
        # 生成请求 ID
        request_id = str(uuid.uuid4())[:8]
        request.state.request_id = request_id
        
        # 记录请求开始
        start_time = time.time()
        logger.info(f"[{request_id}] --> {request.method} {request.url.path}")
        
        # 执行请求
        response = await call_next(request)
        
        # 计算耗时
        duration = (time.time() - start_time) * 1000
        
        # 记录请求结束
        logger.info(
            f"[{request_id}] <-- {response.status_code} "
            f"({duration:.2f}ms)"
        )
        
        # 添加请求 ID 到响应头
        response.headers["X-Request-ID"] = request_id
        
        return response

5.1 SSE 路由

# routers/stream.py

from fastapi import APIRouter, Request
from sse_starlette.sse import EventSourceResponse
import asyncio

router = APIRouter(prefix="/stream", tags=["Stream"])

@router.get("/executions/{execution_id}")
async def execution_stream(execution_id: str, request: Request):
    """订阅执行状态更新流"""
    
    async def event_generator():
        # 连接 Redis Pub/Sub
        pubsub = redis_client.pubsub()
        await pubsub.subscribe(f"execution:{execution_id}")
        
        try:
            while True:
                # 检查客户端是否断开
                if await request.is_disconnected():
                    break
                
                message = await pubsub.get_message(ignore_subscribe_messages=True)
                if message:
                    yield {
                        "event": message["data"]["type"],
                        "data": message["data"]["payload"]
                    }
                
                await asyncio.sleep(0.1)
        finally:
            await pubsub.unsubscribe(f"execution:{execution_id}")
    
    return EventSourceResponse(event_generator())

import redis.asyncio as redis
import json

class EventPublisher:
    def __init__(self, redis_client: redis.Redis):
        self.redis = redis_client
    
    async def publish_execution_event(
        self, 
        execution_id: str, 
        event_type: str, 
        payload: dict
    ):

        channel = f"execution:{execution_id}"
        message = json.dumps({

            "payload": payload
        })

    async def publish_step_completed(self, execution_id: str, step: dict):
        await self.publish_execution_event(
            execution_id,

        )

        await self.publish_execution_event(

# engines/base.py

from abc import ABC, abstractmethod
import httpx

class EngineAdapter(ABC):
    def __init__(self, base_url: str, timeout: int = 30):
        self.base_url = base_url
        self.timeout = timeout
        self.client = httpx.AsyncClient(
            base_url=base_url,
            timeout=timeout
        )
    
    @abstractmethod
    async def execute(self, request: dict) -> dict:
        pass
    
    async def health_check(self) -> bool:
        try:
            response = await self.client.get("/health")
            return response.status_code == 200
        except Exception:
            return False

from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse

app = FastAPI(title="重明 API 网关", version="2.0")

@app.exception_handler(APIException)

            "message": exc.message,

            "code": "INTERNAL_ERROR",
            "message": "An internal error occurred",

    overall = "healthy" if all(v == True or v == "healthy" for v in checks.values()) else "degraded"
    
    return {
        "status": overall,
        "checks": checks
    }

10.2 Prometheus 指标

# middleware/metrics.py

from prometheus_client import Counter, Histogram, generate_latest
from fastapi import Request, Response

REQUEST_COUNT = Counter(
    "http_requests_total",
    "Total HTTP requests",
    ["method", "endpoint", "status"]
)

REQUEST_LATENCY = Histogram(
    "http_request_duration_seconds",
    "HTTP request latency",
    ["method", "endpoint"]
)

class MetricsMiddleware:
    async def __call__(self, request: Request, call_next):
        start_time = time.time()
        
        response = await call_next(request)
        
        duration = time.time() - start_time
        
        # 记录指标
        REQUEST_COUNT.labels(
            method=request.method,
            endpoint=request.url.path,
            status=response.status_code
        ).inc()
        
        REQUEST_LATENCY.labels(
            method=request.method,
            endpoint=request.url.path
        ).observe(duration)
        
        return response

# 指标导出端点
@app.get("/metrics")
async def metrics():
    return Response(content=generate_latest(), media_type="text/plain")
