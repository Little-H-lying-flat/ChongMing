1. 系统概述 (System Overview)

1.1 定义与愿景

Celery 任务调度模块 (Task Scheduler Module) 是重明架构的异步执行基础设施，负责管理长时间运行的测试任务、分布式 Worker 调度、任务优先级队列和执行状态追踪。它像一个智能调度中心，协调各执行引擎的并行工作。

其核心设计哲学是 "Async-First, Scale-Out" (异步优先，横向扩展)：
- 异步解耦：前端请求与后端执行解耦
- 分布式调度：多 Worker 并行执行
- 优先级队列：P0/P1/P2 任务分级处理
- 弹性伸缩：按负载动态调整 Worker 数量

1.2 核心价值

1. **长任务支持** - 支持数小时的压测/回归执行
2. **资源隔离** - 不同引擎的任务独立队列
3. **失败重试** - 自动重试 + 指数退避
4. **实时状态** - WebSocket/SSE 推送执行进度

3. 数据模型设计

3.1 核心数据结构

```python
from dataclasses import dataclass
from enum import Enum
from typing import Optional, Dict, Any, List
from datetime import datetime

class TaskStatus(Enum):
    PENDING = "pending"       # 等待执行
    STARTED = "started"       # 开始执行
    PROGRESS = "progress"     # 执行中 (有进度更新)
    SUCCESS = "success"       # 执行成功
    FAILURE = "failure"       # 执行失败
    RETRY = "retry"           # 重试中
    REVOKED = "revoked"       # 已取消

class TaskPriority(Enum):
    HIGH = 0      # P0 - 紧急
    NORMAL = 5    # P1 - 正常
    LOW = 9       # P2 - 低优先级

class TaskType(Enum):
    UI_TEST = "ui_test"           # 右瞳引擎 UI 测试
    API_TEST = "api_test"         # 左瞳引擎 API 测试
    PERF_TEST = "perf_test"       # 涡轮引擎性能测试
    DESIGN_GEN = "design_gen"     # 神经设计层用例生成
    VRT_TEST = "vrt_test"         # VRT 视觉回归测试
    COMPILE = "compile"           # 凤凰涅槃层脚本编译
    ANALYSIS = "analysis"         # 缺陷分析

@dataclass
class TaskInfo:
    """任务信息"""
    task_id: str
    task_type: TaskType
    priority: TaskPriority
    status: TaskStatus
    args: Dict[str, Any]
    result: Optional[Any]
    created_at: datetime
    started_at: Optional[datetime]
    completed_at: Optional[datetime]
    progress: float         # 0.0 - 1.0
    progress_message: str
    retries: int
    worker_id: Optional[str]
    parent_task_id: Optional[str]  # 父任务 (用于任务链)

@dataclass
class WorkerInfo:
    """Worker 信息"""
    worker_id: str
    hostname: str
    status: str             # online, offline, busy
    task_types: List[str]   # 支持的任务类型
    concurrency: int
    current_tasks: List[str]
    stats: Dict[str, Any]
```

5. 核心任务定义

5.1 测试执行任务

```python
# tasks/execution.py
from celery import shared_task, current_task
from celery.exceptions import SoftTimeLimitExceeded
import asyncio

@shared_task(
    bind=True,
    name="tasks.run_ui_test",
    queue="ui",
    max_retries=3,
    soft_time_limit=3600,
    time_limit=3700,
    acks_late=True
)
def run_ui_test(self, tc_ir: dict, config: dict) -> dict:
    """执行 UI 测试任务 (右瞳引擎)"""
    
    try:
        # 更新进度
        self.update_state(
            state="PROGRESS",
            meta={"progress": 0.0, "message": "初始化右瞳引擎..."}
        )
        
        # 初始化引擎
        from engines.right_pupil import RightPupilEngine
        engine = RightPupilEngine(config)
        
        # 执行测试
        total_steps = len(tc_ir.get("steps", []))
        for i, step in enumerate(tc_ir["steps"]):
            # 进度更新
            progress = (i + 1) / total_steps
            self.update_state(
                state="PROGRESS",
                meta={
                    "progress": progress,
                    "message": f"执行步骤 {i+1}/{total_steps}: {step.get('action')}"
                }
            )
            
            # 执行步骤
            result = asyncio.run(engine.execute_step(step))
            
            if not result.success:
                # 触发自愈
                healed = asyncio.run(engine.try_heal(step, result.error))
                if not healed:
                    return {
                        "status": "failed",
                        "step": i,
                        "error": result.error
                    }
        
        return {
            "status": "passed",
            "trace_log": engine.get_trace_log()
        }
        
    except SoftTimeLimitExceeded:
        # 软超时，保存进度后退出
        return {
            "status": "timeout",
            "progress": self.request.meta.get("progress", 0)
        }
    except Exception as e:
        # 自动重试
        self.retry(exc=e, countdown=60 * (2 ** self.request.retries))


@shared_task(
    bind=True,
    name="tasks.run_api_test",
    queue="api",
    max_retries=3
)
def run_api_test(self, api_ir_chain: list, config: dict) -> dict:
    """执行 API 测试任务 (左瞳引擎)"""
    
    self.update_state(state="PROGRESS", meta={"progress": 0.0, "message": "初始化左瞳引擎..."})
    
    from engines.left_pupil import LeftPupilEngine
    engine = LeftPupilEngine(config)
    
    results = []
    for i, api_ir in enumerate(api_ir_chain):
        progress = (i + 1) / len(api_ir_chain)
        self.update_state(
            state="PROGRESS",
            meta={"progress": progress, "message": f"调用 API: {api_ir['name']}"}
        )
        
        result = asyncio.run(engine.execute(api_ir))
        results.append(result)
    
    return {"status": "completed", "results": results}


@shared_task(
    bind=True,
    name="tasks.run_perf_test",
    queue="turbo",
    time_limit=3600 * 8  # 8 小时
)
def run_perf_test(self, config: dict) -> dict:
    """执行性能测试任务 (涡轮引擎)"""
    
    self.update_state(state="PROGRESS", meta={"message": "启动 Locust 集群..."})
    
    from engines.turbo import TurboEngine
    engine = TurboEngine(config)
    
    # 启动压测
    result = asyncio.run(engine.run_load_test(config))
    
    return {
        "status": "completed",
        "report_url": result.report_url,
        "metrics": result.metrics
    }
```

# tasks/chains.py
from celery import chain, group, chord

def create_full_regression_chain(tc_list: list, config: dict):
    """创建完整回归测试任务链"""
    
    # 1. 首先生成用例 (如果需要)
    generate_task = generate_testcase.s(config.get("prd_content"))
    
    # 2. 并行执行 UI 和 API 测试
    ui_tests = group([
        run_ui_test.s(tc, config) 
        for tc in tc_list if tc["mode"] == "UI"
    ])
    
    api_tests = group([
        run_api_test.s([tc], config) 
        for tc in tc_list if tc["mode"] == "API"
    ])
    
    # 3. 测试完成后生成报告
    report_task = generate_report.s()
    
    # 4. 编排任务链
    workflow = chain(
        group(ui_tests, api_tests),  # 并行执行
        compile_scripts.s(),          # 编译脚本
        report_task                   # 生成报告
    )
    
    return workflow


def create_priority_execution(tc_list: list, config: dict):
    """按优先级执行任务"""
    
    # 按优先级分组
    p0_tasks = [tc for tc in tc_list if tc.get("priority") == "P0"]
    p1_tasks = [tc for tc in tc_list if tc.get("priority") == "P1"]
    p2_tasks = [tc for tc in tc_list if tc.get("priority") == "P2"]
    
    # P0 立即执行，P1/P2 排队
    return chain(
        group([run_ui_test.s(tc, config).set(priority=0) for tc in p0_tasks]),
        group([run_ui_test.s(tc, config).set(priority=5) for tc in p1_tasks]),
        group([run_ui_test.s(tc, config).set(priority=9) for tc in p2_tasks])
    )

6. 进度追踪与 SSE 推送

6.1 实时状态推送

```python
# services/task_tracker.py
from fastapi import APIRouter
from sse_starlette.sse import EventSourceResponse
import asyncio
import json

router = APIRouter()

class TaskTracker:
    """任务状态追踪器"""
    
    def __init__(self, redis_client):
        self.redis = redis_client
        self.pubsub = redis_client.pubsub()
    
    async def track_task(self, task_id: str):
        """追踪任务进度"""
        from celery.result import AsyncResult
        
        while True:
            result = AsyncResult(task_id)
            
            status = {
                "task_id": task_id,
                "status": result.status,
                "progress": 0.0,
                "message": ""
            }
            
            if result.info:
                if isinstance(result.info, dict):
                    status["progress"] = result.info.get("progress", 0)
                    status["message"] = result.info.get("message", "")
                elif isinstance(result.info, Exception):
                    status["error"] = str(result.info)
            
            yield status
            
            if result.ready():
                break
            
            await asyncio.sleep(1)


@router.get("/api/v1/tasks/{task_id}/stream")
async def stream_task_progress(task_id: str):
    """SSE 推送任务进度"""
    
    async def event_generator():
        tracker = TaskTracker(redis_client)
        async for status in tracker.track_task(task_id):
            yield {
                "event": "progress",
                "data": json.dumps(status)
            }
    
    return EventSourceResponse(event_generator())
```

6.2 WebSocket 推送

```python
# services/websocket.py
from fastapi import WebSocket
from typing import Dict, Set

class TaskWebSocketManager:
    """任务 WebSocket 管理器"""
    
    def __init__(self):
        self.connections: Dict[str, Set[WebSocket]] = {}
    
    async def connect(self, task_id: str, websocket: WebSocket):
        await websocket.accept()
        if task_id not in self.connections:
            self.connections[task_id] = set()
        self.connections[task_id].add(websocket)
    
    def disconnect(self, task_id: str, websocket: WebSocket):
        if task_id in self.connections:
            self.connections[task_id].discard(websocket)
    
    async def broadcast_progress(self, task_id: str, data: dict):
        """广播任务进度"""
        if task_id in self.connections:
            for ws in self.connections[task_id]:
                try:
                    await ws.send_json(data)
                except:
                    self.disconnect(task_id, ws)

ws_manager = TaskWebSocketManager()

```

10.2 告警规则

```yaml
# alerting_rules.yaml
groups:
  - name: celery_alerts
    rules:
      - alert: CeleryQueueBacklog
        expr: celery_queue_length > 100
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "Celery queue backlog detected"
          
      - alert: CeleryWorkerDown
        expr: celery_worker_count == 0
        for: 1m
        labels:
          severity: critical
        annotations:
          summary: "All Celery workers are down"
          
      - alert: TaskFailureRate
        expr: rate(celery_task_completed_total{status="failure"}[5m]) > 0.1
        for: 5m
        labels:
          severity: warning
```
