                            1. 模块概述

1.1 定位
--------
神经缓存层 (Neural Cache Layer) 是重明架构中位于智能层 (Intelligence Layer) 
和执行层 (Execution Layer) 之间的性能优化层。通过缓存 AI 推理结果，避免重复
的 OmniParser 调用和 LLM 计算，显著提升执行效率并降低成本。

1.2 核心原则
------------
"Seen Before, Done Before" (见过即做过)

- 页面没变 + 指令没变 = 直接复用上次的坐标
- 意图没变 + API文档没变 = 直接复用上次的 DAG 图

1.3 预期收益
------------
┌─────────────────────────────────────────────────────┐
│                 Neural Cache 收益                   │
├─────────────────────────────────────────────────────┤
│ 执行速度提升:     10x ~ 50x (回归测试场景)          │
│ LLM Token 节省:   50% ~ 80%                         │
│ OmniParser 调用减少: 60% ~ 90%                      │
│ 单步延迟降低:     5-10s → 0.1-0.5s (Cache Hit)      │
└─────────────────────────────────────────────────────┘

5.3 语义相似度判断
------------------
class IntentMatcher:
    """意图匹配器"""
    
    def __init__(self, embedding_model: str = "text-embedding-3-small"):
        self.embedding_model = embedding_model
        self.similarity_threshold = 0.95
    
    async def is_similar_intent(
        self, 
        new_intent: str, 
        cached_intent: str
    ) -> bool:
        """判断两个意图是否语义等价"""
        new_emb = await self.embed(new_intent)
        cached_emb = await self.embed(cached_intent)
        
        similarity = cosine_similarity(new_emb, cached_emb)
        return similarity >= self.similarity_threshold
    
    async def find_matching_cache(
        self,
        intent: str,
        candidates: List[PlanCacheValue]
    ) -> Optional[PlanCacheValue]:
        """从候选缓存中找到匹配项"""
        intent_emb = await self.embed(intent)
        
        for candidate in candidates:
            if cosine_similarity(intent_emb, candidate.intent_emb) >= 0.95:
                return candidate
        
        return None

class CacheWritePolicy:
    """缓存写入策略"""
    
    MIN_CONFIDENCE = 0.8          # 最低置信度
    MIN_EXECUTION_SUCCESS = True  # 必须执行成功
    EXCLUDE_PATTERNS = [          # 不缓存的场景
        "captcha",                # 验证码
        "otp",                    # 一次性密码
        "random",                 # 随机数据
    ]
    
    def should_cache(self, result: CacheableResult) -> bool:
        """判断是否应该缓存"""
        if result.confidence < self.MIN_CONFIDENCE:
            return False
        
        if not result.execution_success:
            return False
        
        for pattern in self.EXCLUDE_PATTERNS:
            if pattern in result.instruction.lower():
                return False
        
        return True

6.3 LRU + 分层存储

class TieredCacheStorage:
    """分层缓存存储"""
    
    def __init__(self):
        self.l1_memory = LRUCache(maxsize=1000)    # 进程内存
        self.l2_redis = RedisClient()              # Redis
        self.l3_s3 = S3Client()                    # 冷存储
    
    async def get(self, key: str) -> Optional[CacheValue]:
        # L1: Memory (最快)
        if value := self.l1_memory.get(key):
            return value
        
        # L2: Redis
        if value := await self.l2_redis.get(key):
            self.l1_memory.set(key, value)  # 回填 L1
            return value
        
        # L3: S3 (最慢，仅冷数据)
        if value := await self.l3_s3.get(key):
            await self.l2_redis.set(key, value)  # 回填 L2
            self.l1_memory.set(key, value)       # 回填 L1
            return value
        
        return None
    
    async def set(self, key: str, value: CacheValue, ttl: int):
        # 同时写入 L1 和 L2
        self.l1_memory.set(key, value)
        await self.l2_redis.set(key, value, ttl=ttl)
        
        # L3 定期由后台任务归档

8.1 Cache Router

class CacheRouter:
    """缓存路由器 - 决策走缓存还是走推理"""
    
    def __init__(
        self,
        visual_cache: VisualCache,
        plan_cache: PlanCache,
        feature_flags: FeatureFlags
    ):
        self.visual_cache = visual_cache
        self.plan_cache = plan_cache
        self.flags = feature_flags
    
    async def route_visual_request(
        self,
        screenshot: bytes,
        instruction: str,
        viewport: str
    ) -> VisualRouteResult:
        """路由视觉定位请求"""
        if not self.flags.enable_visual_cache:
            return VisualRouteResult(use_cache=False)
        
        # 计算缓存键
        page_hash = await self.compute_page_hash(screenshot)
        instruction_norm = self.normalize_instruction(instruction)
        cache_key = f"visual:{page_hash}:{hash(instruction_norm)}:{viewport}"
        
        # 查询缓存
        cached = await self.visual_cache.get(cache_key)
        
        if cached and cached.confidence >= 0.7:
            return VisualRouteResult(
                use_cache=True,
                cached_value=cached,
                cache_key=cache_key
            )
        
        return VisualRouteResult(use_cache=False, cache_key=cache_key)
    
    async def route_plan_request(
        self,
        prd_content: str,
        api_doc: str,
        env: str
    ) -> PlanRouteResult:
        """路由规划请求"""
        if not self.flags.enable_plan_cache:
            return PlanRouteResult(use_cache=False)
        
        # 计算缓存键
        intent_cluster = await self.get_intent_cluster(prd_content)
        api_hash = self.compute_api_hash(api_doc)
        cache_key = f"plan:{intent_cluster}:{api_hash}:{env}"
        
        # 查询缓存
        cached = await self.plan_cache.get(cache_key)
        
        if cached:
            return PlanRouteResult(
                use_cache=True,
                cached_value=cached,
                cache_key=cache_key
            )
        
        return PlanRouteResult(use_cache=False, cache_key=cache_key)

class VisualCache:
    """视觉缓存"""
    
    def __init__(self, storage: TieredCacheStorage):
        self.storage = storage
        self.hasher = ImageHasher()
    
    async def get(self, key: str) -> Optional[VisualCacheValue]:
        """获取缓存"""
        data = await self.storage.get(key)
        if data:
            # 更新命中统计
            await self.update_hit_stats(key)
            return VisualCacheValue.from_json(data)
        return None
    
    async def set(
        self,
        key: str,
        coords: Tuple[int, int],
        locator: str,
        confidence: float,
        screenshot_hash: str,
        page_url: str
    ):
        """写入缓存"""
        value = VisualCacheValue(
            element_coords=coords,
            locator_fallback=locator,
            confidence=confidence,
            screenshot_hash=screenshot_hash,
            page_url=page_url,
            created_at=datetime.now(),
            last_hit_at=datetime.now(),
            hit_count=0
        )
        
        # 根据置信度设置 TTL
        ttl = self.compute_ttl(confidence)
        await self.storage.set(key, value.to_json(), ttl)
    
    def compute_ttl(self, confidence: float) -> int:
        """根据置信度计算 TTL"""
        base_ttl = 3600  # 1 hour
        confidence_factor = confidence ** 2  # 置信度越高，TTL越长
        return int(base_ttl * (1 + confidence_factor * 5))  # 1h ~ 6h

8.3 Perceptual Hash 实现
------------------------
import imagehash
from PIL import Image
import io

class ImageHasher:
    """图像感知哈希"""
    
    def __init__(self, hash_size: int = 16):
        self.hash_size = hash_size
    
    def compute_phash(self, image_bytes: bytes) -> str:
        """计算感知哈希"""
        image = Image.open(io.BytesIO(image_bytes))
        
        # 转换为灰度图
        gray = image.convert('L')
        
        # 计算 pHash
        phash = imagehash.phash(gray, hash_size=self.hash_size)
        
        return str(phash)
    
    def compute_similarity(self, hash1: str, hash2: str) -> float:
        """计算哈希相似度"""
        h1 = imagehash.hex_to_hash(hash1)
        h2 = imagehash.hex_to_hash(hash2)
        
        # 汉明距离
        distance = h1 - h2
        max_distance = self.hash_size ** 2
        
        similarity = 1 - (distance / max_distance)
        return similarity

12.1 与右瞳引擎集成

class RightPupilEngine:
    def __init__(self, cache_router: CacheRouter):
        self.cache_router = cache_router
    
    async def locate_element(self, screenshot, instruction, viewport):
        # 1. 尝试从缓存获取
        route_result = await self.cache_router.route_visual_request(
            screenshot, instruction, viewport
        )
        
        if route_result.use_cache:
            logger.info(f"Visual cache hit: {route_result.cache_key}")
            return route_result.cached_value.element_coords
        
        # 2. 缓存未命中，调用 OmniParser
        coords, confidence = await self.omniparser.detect(screenshot, instruction)
        
        # 3. 结果入缓存
        if confidence >= 0.8:
            await self.cache_router.visual_cache.set(
                key=route_result.cache_key,
                coords=coords,
                confidence=confidence,
                ...
            )
        
        return coords

class NeuralDesignLayer:
    def __init__(self, cache_router: CacheRouter):
        self.cache_router = cache_router
    
    async def generate_test_plan(self, prd_content, api_doc, env):
        # 1. 尝试从缓存获取
        route_result = await self.cache_router.route_plan_request(
            prd_content, api_doc, env
        )
        
        if route_result.use_cache:
            logger.info(f"Plan cache hit: {route_result.cache_key}")
            return route_result.cached_value.dag_graph
        
        # 2. 缓存未命中，调用 LLM 生成
        dag_graph = await self.llm_planner.generate(prd_content, api_doc)
        
        # 3. 结果入缓存
        await self.cache_router.plan_cache.set(
            key=route_result.cache_key,
            dag_graph=dag_graph,
            ...
        )
        
        return dag_graph
