## 1. 测试策略概述

### 1.1 测试金字塔

```
                 ┌─────────┐
                 │   E2E   │  ← 端到端测试 (10%)
                 └────┬────┘
              ┌───────┴───────┐
              │  Integration  │  ← 集成测试 (20%)
              └───────┬───────┘
        ┌─────────────┴─────────────┐
        │         Unit Tests        │  ← 单元测试 (70%)
        └───────────────────────────┘
```

### 1.2 测试类型分布

| 测试类型 | 占比 | 执行频率 | 负责人 |
|----------|------|----------|--------|
| 单元测试 | 70% | 每次提交 | 开发者 |
| 集成测试 | 20% | PR 合并 | 开发者/QA |
| E2E 测试 | 10% | 每日/发布 | QA |

#### 2.2.3 测试用例示例
```python
import pytest
from unittest.mock import AsyncMock, patch

from app.services.design.prd_parser import PRDParser
from app.schemas.prd import PRDParseResult


class TestPRDParser:
    """PRD 解析器测试类"""

    @pytest.fixture
    def parser(self):
        return PRDParser()

    @pytest.fixture
    def sample_prd(self):
        return """
        # 登录功能
        ## 功能描述
        用户可以通过用户名和密码登录系统
        ## 边界条件
        - 用户名长度 4-20 字符
        - 密码长度 8-32 字符
        """

    @pytest.mark.asyncio
    async def test_parse_markdown_prd(self, parser, sample_prd):
        """测试 Markdown 格式 PRD 解析"""
        result = await parser.parse(sample_prd, format="markdown")
        
        assert isinstance(result, PRDParseResult)
        assert "登录功能" in result.modules
        assert len(result.scenarios) > 0

    @pytest.mark.asyncio
    async def test_parse_empty_prd_raises_error(self, parser):
        """测试空 PRD 抛出异常"""
        with pytest.raises(ValueError, match="PRD 内容不能为空"):
            await parser.parse("", format="markdown")

    @pytest.mark.asyncio
    @patch("app.services.design.prd_parser.LLMClient")
    async def test_llm_extraction(self, mock_llm, parser, sample_prd):
        """测试 LLM 调用"""
        mock_llm.return_value.extract = AsyncMock(
            return_value={"modules": ["登录"]}
        )
        
        result = await parser.parse(sample_prd, format="markdown")
        
        mock_llm.return_value.extract.assert_called_once()
```

### 2.3 TypeScript 单元测试

#### 2.3.1 测试框架

- **@testing-library/react** - React 测试
- **MSW** - API Mock

#### 2.3.2 测试用例示例
```typescript
import { describe, it, expect, vi } from 'vitest';
import { render, screen, fireEvent } from '@testing-library/react';
import { TestCaseList } from '@/components/TestCaseList';

describe('TestCaseList', () => {
  it('should render test cases', async () => {
    const mockTestCases = [
      { id: '1', name: '登录测试' },
      { id: '2', name: '注册测试' },
    ];

    render(<TestCaseList testCases={mockTestCases} />);

    expect(screen.getByText('登录测试')).toBeInTheDocument();
    expect(screen.getByText('注册测试')).toBeInTheDocument();
  });

  it('should call onSelect when clicked', async () => {
    const handleSelect = vi.fn();
    const mockTestCases = [{ id: '1', name: '登录测试' }];

    render(<TestCaseList testCases={mockTestCases} onSelect={handleSelect} />);
    
    fireEvent.click(screen.getByText('登录测试'));

    expect(handleSelect).toHaveBeenCalledWith(mockTestCases[0]);
  });
});
```

### 3.3 集成测试示例

```python
import pytest
from httpx import AsyncClient
from sqlalchemy.ext.asyncio import AsyncSession

from app.main import app
from app.core.database import get_db


@pytest.fixture
async def client():
    async with AsyncClient(app=app, base_url="http://test") as ac:
        yield ac


@pytest.fixture
async def db_session():
    async with AsyncSession() as session:
        yield session
        await session.rollback()


class TestDesignAPI:
    """设计层 API 集成测试"""

    @pytest.mark.integration
    async def test_generate_test_cases_flow(self, client, db_session):
        """测试完整的用例生成流程"""
        # 1. 解析 PRD
        prd_response = await client.post(
            "/api/v1/design/parse-prd",
            json={"content": "# 登录功能...", "format": "markdown"}
        )
        assert prd_response.status_code == 200
        prd_id = prd_response.json()["prd_id"]

        # 2. 生成用例
        gen_response = await client.post(
            "/api/v1/design/generate",
            json={"prd_id": prd_id, "coverage_strategy": "smoke"}
        )
        assert gen_response.status_code == 200
        tc_list = gen_response.json()["tc_list"]
        
        assert len(tc_list) > 0
        assert all(tc["protocol"] == "TC-IR" for tc in tc_list)
```

import factory

fake = Faker("zh_CN")

class TestCaseFactory(factory.Factory):
    class Meta:
        model = dict

    id = factory.LazyFunction(lambda: f"tc-{fake.uuid4()[:8]}")
    name = factory.LazyFunction(lambda: f"{fake.word()}功能测试")
